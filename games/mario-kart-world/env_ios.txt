# The iPhone app's switches: switch2mac.py ios packs them into the app (dm/env.txt), and switch2mac.py run --phone uses them on the Mac, to try the phone's settings there first.
# On the phone, Documents/env.txt overrides any of them with no rebuild (one KEY=VALUE per line, # starts a comment). docs/ios.md says what each is for.
# Memory statistics every second (memstat.m): Documents/memstat.tsv on the phone
DM_MEMSTAT=memstat.tsv
# The largest block of address space the app can reserve, logged at start: whether LiveContainer passes on the extended virtual addressing entitlement
DM_VA_PROBE=1
# The memory switches measured on the Mac with run --phone (part D: footprint median 11.6 GB over title, race and free roam instead of 12.9, 60 fps, the picture unchanged; docs/ios.md):
# textures deswizzled on the GPU from a copy of their storage, so their pools stay out of the residency set, which iOS wires whole
DM_NVN_DESWZ_GPU=2
# large sparse textures the game draws into get memory only for the 64 KB tiles it maps (Mario Kart World: 10 such textures, 3.1 GB when made whole)
DM_NVN_SPARSE_RT=1
# the terrain's sparse tiles the game unmaps give their memory back, and its tiles over the same physical tile share one Metal tile (heaps 4.4 GB -> 2.0 GB by the end of free roam)
DM_NVN_SPARSE_FREE=1
DM_NVN_SPARSE_SHARE=1
